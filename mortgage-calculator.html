// 等待DOM加载完成 document.addEventListener('DOMContentLoaded', function() { // 初始化Chart.js let paymentChart = null; // 获取DOM元素 const readMoreBtn = document.getElementById('read-more'); const fullContent = document.getElementById('full-content'); const toolCards = document.querySelectorAll('.tool-card'); const calculateBtn = document.getElementById('calculate-mortgage'); const calculationResults = document.getElementById('calculation-results'); // 阅读全文按钮功能 if (readMoreBtn && fullContent) { readMoreBtn.addEventListener('click', function() { fullContent.classList.toggle('expanded'); // 更新按钮文本和图标 if (fullContent.classList.contains('expanded')) { this.innerHTML = ' 收起全文'; this.classList.add('expanded'); } else { this.innerHTML = ' 阅读全文'; this.classList.remove('expanded'); } }); } // 左侧工具按钮功能 toolCards.forEach(card => { card.addEventListener('click', function() { // 移除所有工具的active状态 toolCards.forEach(c => c.classList.remove('active')); // 给当前点击的工具添加active状态 this.classList.add('active'); const toolId = this.id; console.log(`工具按钮被点击: ${toolId}`); // 根据按钮ID执行不同功能 switch(toolId) { case 'exchange-rate': alert('汇率转换功能正在开发中...'); break; case 'tax-calc': alert('个税计算功能正在开发中...'); break; case 'mortgage-calc': // 滚动到房贷计算器 document.getElementById('mortgage-calculator').scrollIntoView({ behavior: 'smooth' }); break; case 'history': alert('历史存档功能正在开发中...'); break; } }); }); // 连接输入框和滑块 const loanAmountInput = document.getElementById('loan-amount'); const loanAmountSlider = document.getElementById('loan-amount-slider'); const loanRateInput = document.getElementById('loan-rate'); const loanRateSlider = document.getElementById('loan-rate-slider'); if (loanAmountInput && loanAmountSlider) { loanAmountInput.addEventListener('input', function() { loanAmountSlider.value = this.value; }); loanAmountSlider.addEventListener('input', function() { loanAmountInput.value = this.value; }); } if (loanRateInput && loanRateSlider) { loanRateInput.addEventListener('input', function() { loanRateSlider.value = this.value; }); loanRateSlider.addEventListener('input', function() { loanRateInput.value = this.value; }); } // 房贷计算功能 if (calculateBtn) { calculateBtn.addEventListener('click', calculateMortgage); } // 房贷计算函数 function calculateMortgage() { // 获取输入值 const loanAmount = parseFloat(loanAmountInput.value) * 10000; // 转换为元 const annualRate = parseFloat(loanRateInput.value) / 100; // 转换为小数 const monthlyRate = annualRate / 12; // 获取贷款年限 const loanTermRadios = document.querySelectorAll('input[name="loan-term"]:checked'); let loanTermYears = 30; // 默认30年 if (loanTermRadios.length > 0) { loanTermYears = parseInt(loanTermRadios[0].value); } const loanTermMonths = loanTermYears * 12; // 获取还款方式 const repaymentRadios = document.querySelectorAll('input[name="repayment-type"]:checked'); let repaymentType = 'equal-principal-interest'; // 默认等额本息 if (repaymentRadios.length > 0) { repaymentType = repaymentRadios[0].value; } let monthlyPayment = 0; let totalInterest = 0; let totalPayment = 0; let interestRatio = 0; // 根据还款方式计算 if (repaymentType === 'equal-principal-interest') { // 等额本息计算 monthlyPayment = loanAmount * monthlyRate * Math.pow(1 + monthlyRate, loanTermMonths) / (Math.pow(1 + monthlyRate, loanTermMonths) - 1); totalPayment = monthlyPayment * loanTermMonths; totalInterest = totalPayment - loanAmount; } else { // 等额本金计算 const monthlyPrincipal = loanAmount / loanTermMonths; let remainingPrincipal = loanAmount; totalPayment = 0; for (let i = 1; i <= loanTermMonths; i++) { const monthlyInterest = remainingPrincipal * monthlyRate; const currentMonthlyPayment = monthlyPrincipal + monthlyInterest; totalPayment += currentMonthlyPayment; remainingPrincipal -= monthlyPrincipal; } totalInterest = totalPayment - loanAmount; // 等额本金第一个月的月供 monthlyPayment = loanAmount / loanTermMonths + loanAmount * monthlyRate; } interestRatio = (totalInterest / totalPayment) * 100; // 更新显示结果 document.getElementById('monthly-payment').textContent = `¥${monthlyPayment.toFixed(2)}`; document.getElementById('total-interest').textContent = `¥${totalInterest.toFixed(2)}`; document.getElementById('total-payment').textContent = `¥${totalPayment.toFixed(2)}`; document.getElementById('interest-ratio').textContent = `${interestRatio.toFixed(2)}%`; // 显示结果区域 calculationResults.style.display = 'block'; // 更新图表 updateChart(loanAmount, totalInterest, repaymentType); } // 更新图表函数 function updateChart(principal, interest, repaymentType) { const ctx = document.getElementById('payment-chart').getContext('2d'); // 如果已有图表实例，先销毁 if (paymentChart) { paymentChart.destroy(); } // 创建新图表 paymentChart = new Chart(ctx, { type: 'doughnut', data: { labels: ['贷款本金', '总利息'], datasets: [{ data: [principal, interest], backgroundColor: [ '#1a3a8f', '#ff6b6b' ], borderWidth: 1 }] }, options: { responsive: true, maintainAspectRatio: false, plugins: { legend: { position: 'bottom', }, tooltip: { callbacks: { label: function(context) { let label = context.label || ''; if (label) { label += ': '; } const value = context.raw; label += `¥${(value/10000).toFixed(2)}万元`; return label; } } } } } }); } // 页面加载后自动计算一次 calculateMortgage(); });
